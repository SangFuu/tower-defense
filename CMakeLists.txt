cmake_minimum_required(VERSION 3.16)
project(TowerDefense VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SDL2_ROOT "" CACHE PATH "Dossier SDL2 avec include/SDL2, lib et bin (MinGW Windows)")
if(SDL2_ROOT AND MINGW)
  find_path(SDL2_INCLUDE_DIR SDL.h PATHS "${SDL2_ROOT}/include/SDL2" NO_DEFAULT_PATH REQUIRED)
  find_library(SDL2_IMPORT_LIBRARY NAMES SDL2.dll SDL2 PATHS "${SDL2_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(SDL2::SDL2 SHARED IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_IMPLIB "${SDL2_IMPORT_LIBRARY}"
    IMPORTED_LOCATION "${SDL2_ROOT}/bin/SDL2.dll" INTERFACE_INCLUDE_DIRECTORIES "${SDL2_INCLUDE_DIR}")
else()
  find_package(SDL2 CONFIG REQUIRED)
endif()
add_library(tower_core towerdefend.c maSDL.c)
target_include_directories(tower_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tower_core PUBLIC SDL_MAIN_HANDLED)
target_link_libraries(tower_core PUBLIC SDL2::SDL2)
if(NOT MSVC)
  target_compile_options(tower_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(tower-defense main.c)
target_link_libraries(tower-defense PRIVATE tower_core)
if(NOT MSVC)
  target_compile_options(tower-defense PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_custom_command(TARGET tower-defense POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:tower-defense>/data")
if(WIN32 AND SDL2_ROOT)
  add_custom_command(TARGET tower-defense POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SDL2_ROOT}/bin/SDL2.dll" "$<TARGET_FILE_DIR:tower-defense>/SDL2.dll")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_game tests/test_game.c)
  target_link_libraries(test_game PRIVATE tower_core)
  if(NOT MSVC)
    target_compile_options(test_game PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME logique-et-sauvegardes COMMAND test_game)
  add_test(NAME affichage-et-fermeture COMMAND tower-defense --seed 42 --smoke-test 120)
  set_tests_properties(affichage-et-fermeture PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 15)
  set_tests_properties(logique-et-sauvegardes PROPERTIES TIMEOUT 30)
endif()
